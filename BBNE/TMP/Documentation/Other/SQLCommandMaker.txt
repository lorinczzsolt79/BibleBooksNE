
BibleInfos class parts

        /* SQLite data struct
         *      byte number
         *      text longBibleName
         *      text shortBibleName
         *      text otherBibleName
         *      text language
         *      text description
         *      text url
         *      text version
         *      text myversion
         */

        public string InsertIntoCommand()
        {
            StringBuilder sb = new StringBuilder();
            sb.Append("INSERT INTO BibleInfos (number, longBibleName, shortBibleName, otherBibleName, language, description, url, version, myversion) VALUES");
            sb.Append(Fields());
            sb.Append("; ");
            return sb.ToString();
        }

        private string Fields()
        {
            return "(" +
                Number + ", '" +
                LongBibleName + "', '" +
                ShortBibleName + "', '" +
                OtherBibleName + "', '" +
                Language + "', '" +
                Description + "', '" +
                Url + "', '" +
                Version + "', '" +
                MyVersion + "'" +
                ")";
        }
